/** * QR在庫管理・レジ バックエンド (Google Apps Script) * 初期設定: setup() を1回実行 → 管理者PINとスプレッドシートが用意される */ const ITEM_HEADERS = ['ID', '種別', '名前', '写真URL', '個数', '保管場所', '単価', 'メモ', '更新日時', 'QR名']; const SALE_HEADERS = ['日時', '伝票ID', 'ID', '名前', '数量', '単価', '小計']; const PUBLIC_FNS = ['getItems', 'getItem', 'checkout']; const ADMIN_FNS = ['saveItem', 'deleteItem', 'uploadPhoto', 'getSales', 'adjustStock']; function setup() { const props = PropertiesService.getScriptProperties(); if (!props.getProperty('ADMIN_PIN')) props.setProperty('ADMIN_PIN', '1234'); // 必ず変更する const ss = getSS_(); sheet_(ss, 'items', ITEM_HEADERS); sheet_(ss, 'sales', SALE_HEADERS); if (!props.getProperty('PHOTO_FOLDER')) { const f = DriveApp.createFolder('QR在庫管理_写真'); props.setProperty('PHOTO_FOLDER', f.getId()); } } function getSS_() { const props = PropertiesService.getScriptProperties(); let id = props.getProperty('SS_ID'); if (id) return SpreadsheetApp.openById(id); const active = SpreadsheetApp.getActiveSpreadsheet(); // スプレッドシートに紐付けた場合 if (active) { props.setProperty('SS_ID', active.getId()); return active; } const ss = SpreadsheetApp.create('QR在庫管理DB'); props.setProperty('SS_ID', ss.getId()); return ss; } function sheet_(ss, name, headers) { let sh = ss.getSheetByName(name); if (!sh) { sh = ss.insertSheet(name); sh.getRange(1, 1, 1, headers.length).setValues([headers]).setFontWeight('bold'); sh.setFrozenRows(1); } else if (sh.getLastColumn() < headers.length) { // 既存シートに新しい列(QR名など)を足す sh.getRange(1, 1, 1, headers.length).setValues([headers]).setFontWeight('bold'); } return sh; } // ---- Webアプリ入口 ---- function doGet() { try { return HtmlService.createHtmlOutputFromFile('index') .setTitle('QRレジ・備品管理') .addMetaTag('viewport', 'width=device-width, initial-scale=1') .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL); } catch (e) { // GitHub Pages で画面を配信する構成では、GAS側に index.html は不要（APIとして動作） return ContentService.createTextOutput('QR在庫管理 API is running'); } } // GitHub Pages など外部ホストから呼ぶ場合用 (Content-Type: text/plain で JSON を送る) function doPost(e) { let out; try { const req = JSON.parse(e.postData.contents); out = { ok: true, data: dispatch(req.fn, req.args || [], req.pin) }; } catch (err) { out = { ok: false, error: String(err.message || err) }; } return ContentService.createTextOutput(JSON.stringify(out)).setMimeType(ContentService.MimeType.JSON); } // google.script.run からはこの1関数だけを呼ぶ function dispatch(fn, args, pin) { if (ADMIN_FNS.indexOf(fn) >= 0) requirePin_(pin); else if (PUBLIC_FNS.indexOf(fn) < 0 && fn !== 'verifyPin') throw new Error('不明な操作です'); if (fn === 'verifyPin') return verifyPin_(pin); return this[fn + '_'].apply(null, args); } function verifyPin_(pin) { return String(pin) === PropertiesService.getScriptProperties().getProperty('ADMIN_PIN'); } function requirePin_(pin) { if (!verifyPin_(pin)) throw new Error('管理者PINが違います'); } // ---- データ操作 ---- function readItems_() { const sh = sheet_(getSS_(), 'items', ITEM_HEADERS); const rows = sh.getDataRange().getValues().slice(1); return rows.filter(r => r[0] !== '').map((r, i) => ({ row: i + 2, id: String(r[0]), type: r[1], name: r[2], photo: r[3], qty: Number(r[4]) || 0, place: r[5], price: Number(r[6]) || 0, memo: r[7], label: r[9] || '', updated: r[8] ? Utilities.formatDate(new Date(r[8]), 'Asia/Tokyo', 'M/d HH:mm') : '' })); } function getItems_() { return readItems_(); } function getItem_(id) { return readItems_().filter(x => x.id === id)[0] || null; } function saveItem_(item) { const lock = LockService.getScriptLock(); lock.waitLock(20000); try { const sh = sheet_(getSS_(), 'items', ITEM_HEADERS); const existing = item.id ? getItem_(item.id) : null; const id = item.id || ((item.type === 'equipment' ? 'E-' : 'F-') + Utilities.getUuid().slice(0, 8)); const row = [id, item.type || 'festival', item.name || '', item.photo || '', Number(item.qty) || 0, item.place || '', Number(item.price) || 0, item.memo || '', new Date(), item.label || '']; if (existing) sh.getRange(existing.row, 1, 1, row.length).setValues([row]); else sh.appendRow(row); return getItem_(id); } finally { lock.releaseLock(); } } function deleteItem_(id) { const it = getItem_(id); if (it) sheet_(getSS_(), 'items', ITEM_HEADERS).deleteRow(it.row); return true; } function adjustStock_(id, delta, memo) { const it = getItem_(id); if (!it) throw new Error('未登録の商品です'); it.qty = Math.max(0, it.qty + Number(delta)); if (memo) it.memo = memo; return saveItem_(it); } /** cart: [{id, qty}] 在庫を一括で引き落とし、売上を記録する */ function checkout_(cart) { const lock = LockService.getScriptLock(); lock.waitLock(20000); try { const ss = getSS_(); const items = sheet_(ss, 'items', ITEM_HEADERS); const sales = sheet_(ss, 'sales', SALE_HEADERS); const all = readItems_(); const slip = Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyyMMddHHmmss') + '-' + Utilities.getUuid().slice(0, 4); const now = new Date(); const logs = []; let total = 0; // 先に全部検証してから書き込む const plan = cart.map(c => { const it = all.filter(x => x.id === c.id)[0]; if (!it) throw new Error('未登録の商品が含まれています: ' + c.id); const q = Math.floor(Number(c.qty)); if (!(q > 0)) throw new Error('数量が不正です: ' + it.name); if (it.qty < q) throw new Error('在庫不足: ' + it.name + '（残り' + it.qty + '個）'); return { it: it, q: q }; }); plan.forEach(p => { items.getRange(p.it.row, 5).setValue(p.it.qty - p.q); items.getRange(p.it.row, 9).setValue(now); total += p.it.price * p.q; logs.push([now, slip, p.it.id, p.it.name, p.q, p.it.price, p.it.price * p.q]); }); if (logs.length) sales.getRange(sales.getLastRow() + 1, 1, logs.length, SALE_HEADERS.length).setValues(logs); return { slip: slip, total: total }; } finally { lock.releaseLock(); } } function uploadPhoto_(base64, mime, name) { const folder = DriveApp.getFolderById(PropertiesService.getScriptProperties().getProperty('PHOTO_FOLDER')); const blob = Utilities.newBlob(Utilities.base64Decode(base64), mime, name || 'photo.jpg'); const file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); return 'https://drive.google.com/thumbnail?id=' + file.getId() + '&sz=w600'; } function getSales_() { const sh = sheet_(getSS_(), 'sales', SALE_HEADERS); return sh.getDataRange().getValues().slice(1).map(r => [ Utilities.formatDate(new Date(r[0]), 'Asia/Tokyo', 'yyyy-MM-dd HH:mm:ss'), r[1], r[2], r[3], r[4], r[5], r[6] ]); }
